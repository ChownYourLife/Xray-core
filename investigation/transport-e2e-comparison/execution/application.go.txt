func must(err error) { if err!=nil { panic(err) } }
func exact(c io.Reader,n int) []byte { b:=make([]byte,n);_,e:=io.ReadFull(c,b);must(e);return b }
func put(c io.Writer,b []byte) { for len(b)>0 { n,e:=c.Write(b);must(e);if n==0 { panic("short write") }; b=b[n:] } }
func number(n uint64) []byte { b:=make([]byte,8);binary.BigEndian.PutUint64(b,n);return b }

const window=32
const udpPayload=1200
func dataPacket(seq uint32) []byte {b:=make([]byte,udpPayload+5);b[0]='P';binary.BigEndian.PutUint32(b[1:5],seq);copy(b[5:],block[:udpPayload]);return b}
type udpState struct {base uint32;seen uint32;count uint64;start time.Time}
